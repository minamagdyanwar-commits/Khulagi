1: تُعتبَرُ القدَّاساتُ الثلاثةُ المُستعمَلةُ الآنَ في الكنيسةِ القبطيَّةِ الأرثوذكسيَّةِ مستقرَّةً منذُ القرنِ الثاني عشرَ من عهدِ البابا غبريالَ بنِ تريكَ، ويتجلَّى ذلكَ في القانونِ السادسِ والعشرينَ مِن قوانينِهِ، حينما منعَ الصلاةَ بغيرِ هؤلاءِ الثلاثةِ. (مخطوط ٢٥١ عربي بالمكتبةِ الأهليةِ بباريس، بتاريخ ١٣٥٣م) وقد ذكرَ أبو البركاتِ بنُ كبرٍ (ق١٣–١٤) في البابِ السابعَ عشرَ، مناسباتِ هذهِ القدَّاساتِ حيثُ قالَ: "والذي استقرَّ في البيعةِ القبطيةِ مِنَ القدَّاساتِ ثلاثةٌ، أحدُها وهو المُستعمَلُ في سائرِ الأيامِ صومًا وفطرًا، قدَّاسُ القدِّيسِ باسيليوسَ أسقفِ قيساريةَ قبادوقيةَ، والثاني وجرتْ عادةُ المصريينَ أنْ لا يُقدِّسوا بهِ إلا في الصومِ الكبيرِ وشهرِ كيهكَ، قدَّاسُ مرقسَ الذي كمَّلهُ كيرلسُ، والثالثُ وهو خصيصٌ بأيامِ الأعيادِ الإلهيةِ والفرحِ وهو قدَّاسُ القدِّيسِ إغريغوريوسَ" (مخطوط ٢٠٣ عربي بالمكتبةِ الأهليةِ بباريس). (الناشر).
2: خر٣٠: ١٧–٢١.
3: أصلُ الاحتراسِ الذي نصفُهُ قبلَ التناولِ هو ما ذُكرَ في البابِ الثالثَ عشرَ مِنَ المجموعِ الصفويِّ المُختصِّ بالقربانِ، لأنَّهُ في البابِ المذكورِ في عددِ ١٧ و١٨ قيلَ: (قالَ الرُّسلُ ٤٣ مِنْ ٧١ وبُطرسُ): لا يتناولُ أحدُ القربانِ إلا وهو صائمٌ نقيٌّ، ومَنْ أفطرَ مِنَ المؤمنينَ والمؤمناتِ ثُمَّ تقرَّبَ إنْ كانَ فعلَ ذلكَ تهاونًا بهِ فليُنفَ عن كنيسةِ اللهِ إلى الأبدِ. (وقالَ بوليدس ٢٨ والرُّسلُ ٤٣ مِنْ ٧١) وكُلُّ مؤمنٍ فليجعلْ دأبَهُ أنْ يتناولَ مِنَ الأسرارِ قبلَ أنْ يذوقَ شيئًا لا سيما أيامَ الصومِ. أهـ. فأقولُ إنَّ قولَ القوانينِ: "لا يذوقُ شيئًا" يمنعُ الأكلَ والشربَ كليَّةً حتى ما يكونُ أقلَّ منهما، ولا يمنعُ البصقَ. والعادةُ الجاريةُ في كنائسنا القبطيَّةِ الآنَ هي أنْ يكونَ الاحتراسُ تسعَ ساعاتٍ قبلَ التناولِ، وأمَّا بعدَ التناولِ فالبعضُ يحترسونَ أربعًا وعشرينَ ساعةً لا يبصقونَ ولا يضعونَ في فمِهم شيئًا ويحرجونَهُ مهمَا كانَ حتى إصبعَهم، وأمَّا الأكلُ والشربُ وكلُّ ما يدخلُ الفمَ مِنْ غيرِ خروجٍ فمباحٌ، والبعضُ يكتفونَ بالتي عشرةَ ساعةً بعدَ التناولِ أو تسعِ ساعاتٍ إذا كانوا يتناولونَ بكثرةٍ فيملُّونَ مِنَ الاحترازِ. أهـ (م).
4: وقبلَ لبسِ الكاهنِ التونيةَ يرشُمُها ثلاثةَ رشومٍ وهو يقولُ ϧⲉⲛ ⲫ̀ⲣⲁⲛ و ⲧⲥⲁⲣⲱⲟⲩⲧ̀ إلخ، ثمَّ يلبسُها، وكذلكَ كلُّ كاهنٍ أو شمَّاسٍ أرادَ التناولَ يرشُمُ تونيتَهُ قبلَ لبسِها. فالكاهنُ بيدهِ والشمَّاسُ يُقدِّمُها إلى أحدِ الكهنةِ فيرشُمُها لهُ. وإنِ اشتركَ كاهنانِ في رشمِ تونيةٍ أو جملةِ توانٍ، أو أيضًا في رشمِ أشياءَ غيرِ التواني فإنَّ أحدَهما يرشُمُ الرشمينِ الأولَ والثالثَ ويقولُ ما يخصُّهما والآخرُ يرشُمُ فقطِ الرشمَ الثاني ويقولُ ما يخصُّهُ، كما شُرحَ في رشمِ دَرْجِ البخورِ قبلَ أوشيةِ بخورِ عشيةٍ في وجه ٤٤. وأمَّا إذا كانَ الأبُ البطريركُ أو الأسقفُ حاضرًا فهو الذي يرشُمُ لهم.
5: وتدومُ هذهِ الثلاثةُ موقدةً مِنْ أولِ الصلاةِ إلى آخرِ القدَّاسِ وانتهاءِ التناولِ. وانظرْ أيضًا القوانينَ قبلَ القدَّاسِ باب ١ عدد ٢ و٣ وباب ١٣ عدد ١.
6: انظرِ القوانينَ المذكورةَ باب ١٢ عدد ١٣. ويكونُ الشمَّاسُ قد ملأَ الإبريقَ ماءً وعمَّرَ المجمرةَ بالفحمِ وأشعلَها إذا كانتْ محتاجةً إلى التعميرِ، وعندَ طلوعِ الشمامسةِ إلى الهيكلِ في وقتٍ يسبقونَ مِنْ عتبةِ البابِ برجلِهمُ اليُمنى. وعندَ نزولِهم يسبقونَ برجلِهمُ اليُسرى ويكونُ وجهُهم إلى الشرقِ.
7: وقبلما يحلُّ الآنيةَ وهي مربوطةٌ يرشُمُها ثلاثةَ رشومٍ وهو يقولُ: ϧⲉⲛ ⲫ̀ⲣⲁⲛ و ⲧⲥⲁⲣⲱⲟⲩⲧ̀ إلخ كالعادةِ ثمَّ يحلُّها.
8: العادةُ جاريةٌ الآنَ أنَّهُ أولاً يُصلِّي الشعبُ صلاةَ مزامير الساعاتِ. ويكونُ استعدادُ المذبحِ قبلَها أو في أثنائِها. ويحضرُ القربانُ والخمرُ قبلَ صلاةِ المزامير لئلا يتأخَّرَ أحدُهما فتُبطلَ الصلاةُ وقتًا، ولئلا يحصلَ لأحدِهما عارضٌ فيُبطلَ القدَّاسَ بالكليةِ في ذلكَ اليومِ. (م). تنبيهٌ: اعلمْ أنَّ نُسخًا نادرةً ذكرتْ أنَّ الشعبَ يقولُ ⲁⲗⲗⲏⲗⲟⲩⲓⲁ ⲫⲁⲓ ⲡⲉ ⲡⲓϩⲟⲟⲩ وأختيها هنا في وقتِ استعدادِ المذبحِ كلُّ واحدةٍ في أيامِها. ولم تذكرْها عندَ دورةِ الحملِ. لكنَّ أكثرَ النُّسخِ والجاري في كنائسِنا الآنَ أنَّها تُقالُ في دورةِ الحملِ، فكذلكَ وضعناها في هذا الكتابِ. وأمَّا الخولاجي المطبوعُ برومية سنةَ ١٤٥٢ للشهداءِ فجعلَها تُقالُ مرَّتينِ أي هنا وهناكَ، فكأنَّهُ أخذَ القولينِ معًا. وذلكَ غيرُ موافقٍ. أهـ (م).
9: أع١: ٢٤، ١٥: ٨، مز٤٣: ٢١، لو١٦: ١٥.

! نقاوةُ الكاهنِ
a: يلزمُ الكاهنَ أولاً أنْ يستبرئَ أفكارَهُ وأحوالَهُ داخلاً وخارجًا. ففي الداخلِ (وهو الأهمُّ جدًّا): يجبُ أنْ يكونَ نقيًّا مِنْ كُلِّ زلَّةٍ لئلا يأخذَ دينونةً لنفسِهِ، لقولِ بولسَ الرسولِ: «أيُّ مَنْ أكلَ هذا الخبزَ أو شربَ كأسَ الربِّ بدونِ استحقاقٍ

# 206
a: يكونُ مُجرمًا في جسدِ الربِّ ودمِهِ، ولكنْ ليَمتحِنِ الإنسانُ نفسَهُ وهكذا يأكلُ مِنَ الخبزِ ويشربُ مِنَ الكأسِ، لأنَّ الذي يأكلُ ويشربُ بدونِ استحقاقٍ يأكلُ ويشربُ دينونةً لنفسِهِ غيرَ مُميِّزٍ جسدَ الربِّ، مِنْ أجلِ هذا فيكم كثيرونَ ضعفاءُ ومرضى وكثيرونَ يرقدونَ، لأنَّنا لو كُنَّا حكمنا على أنفسِنا لَمَا حُكِمَ علينا» (١كو١١: ٢٧–٣١) ويكونُ الكاهنُ مُصطلحًا مع الجميعِ، لقولِ سيِّدِنا: «فإنْ قدَّمتَ قُربانَكَ إلى المذبحِ وهناكَ تذكَّرتَ أنَّ لأخيكَ شيئًا عليكَ فاترُكْ هناكَ قُربانَكَ قُدَّامَ المذبحِ واذهبْ أولاً اصطلحْ مع أخيكَ، وحينئذٍ تعالَ وقدِّمْ قُربانَكَ» (مت٥: ٢٣و٢٤) وقولِهِ أيضًا: «ومتى وقفتُم تُصلُّونَ فاغفروا إنْ كانَ لكُم على أحدٍ شيءٌ لكي يغفرَ لكُم أيضًا أبوكُمُ الذي في السماواتِ زلاَّتِكُم. وإنْ لم تغفروا أنتُم لا يغفرْ أبوكُمُ الذي في السماواتِ أيضًا زلاَّتِكُم» (مر١١: ٢٥و٢٦)، فإنْ كانَ الكاهنُ نقيًّا مِنْ كُلِّ خطيةٍ فهو يأخذُ بتناولِهِ مِنْ جسدِ الربِّ ودمِهِ مغفرةَ الخطايا (مت٢٦: ٢٦–٢٨)، وحياةً أبديةً (يو٦: ٥٤).

a: وفي الخارجِ: يجبُ أنْ يكونَ نظيفَ الجسمِ والملابسِ ويغسلَ يديهِ ورجليهِ[^2] (ويكونَ محترسًا أي لم يذقْ شيئًا البتةَ).[^3]

# 207
! بدءُ الاستعدادِ
a: ثُمَّ يلبسُ الكاهنُ بدلةَ الكهنوتِ[^4] وهو يقولُ المزمورَ التاسعَ والعشرينَ (أُعظِّمُكَ يا ربُّ) وبعدَهُ المزمورَ الثانيَ والتسعينَ (الربُّ قد مَلَكَ والجمالَ لَبِسَ) ثُمَّ يُصافحُ إخوتَهُ الكهنةَ ويسألُهم مساعدتَهُ في الطلبِ، ثُمَّ يخضعُ للربِّ أمامَ هيكلِهِ المُقدَّسِ، ويضربُ مطانيةً لإخوتِهِ الكهنةِ وباقي الإكليروسِ. وأمَّا ترتيبُهُ في طلوعِهِ إلى المذبحِ ونزولِهِ منهُ فكما جرتْ بهِ العادةُ مثلَ ما شُرِحَ في رفعِ البخورِ، أنَّهُ إذا طلعَ يسبقُ مِنْ عتبةِ البابِ برجلِهِ اليُمنى ويُقبِّلُ المذبحَ بفمِهِ (أو يسجُدُ فوقَهُ برأسِهِ للهِ)، وإذا نزلَ يسبقُ برجلِهِ اليُسرى ويكونُ وجهُهُ إلى الشرقِ فيطلعُ الآنَ كما شُرحَ، ويكونُ قنديلُ الشرقِ موقدًا، وكذلكَ شمعتانِ موقدتانِ على جانبَيِ المذبحِ على شمعدانَينِ،[^5] والشمَّاسُ يصعدُ ويقفُ أمامَهُ[^6] ثُمَّ إنَّ الكاهنَ يكشفُ المائدةَ مِنَ الإبروسفارينِ، ويضعُ الآنيةَ أمامَهُ ويحلُّها مِنْ رباطِها، (بعدَ تنفيضِ فرشِ المذبحِ بلفافةٍ أو غيرِها أو بعدَ فرشِهِ إنْ كانَ غيرَ مفروشٍ).[^7]

! هنا يبتدئُ الكاهنُ يمسحُ الصينيةَ ويضعُها مكانَها، ثُمَّ يمسحُ الكأسَ ويُفتِّشُ داخلَ الكرسيِّ ويمسحُهُ مِنْ فوقُ ويضعُ الكأسَ فيهِ، وكذلكَ يمسحُ الملعقةَ ويضعُها على الكرسيِّ قبليَّ الكأسِ، ويضعُ اللفائفَ في أماكنِها، ويكونُ قد وضعَ منها لفافةً تحتَ الصينيةِ وغطَّاها بلفافةٍ أخرى وغطَّى الكأسَ أيضًا بلفافةٍ.

# 214
! تقديمُ الحملِ
a: ثُمَّ يُقبِّلُ الكاهنُ المذبحَ بفمِهِ، وعندَ تقديمِ الحملِ[^16] يقفُ في بابِ الهيكلِ مُتجهًا إلى الغربِ ومعهُ لفافةٌ، ويُقدِّمونَ لهُ الحملَ والخمرَ،[^17] فيختارُ الحملَ الذي هو خبزُ التقدمةِ، ويستبرئُ ذلكَ جيدًا، لأنَّهُ مكتوبٌ عن خروفِ فصحِ العهدِ القديمِ أنْ يكونَ حملاً حوليًّا لا عيبَ فيهِ (انظرْ خروج١٢: ٥، ولاويين٢٢: ١٧ إلى ٢٥) ويأخذُ الحملَ بعدَ استبرائِهِ[^18] ويمسحُ ظهرَهُ بسترٍ نظيفٍ أي بلفافةٍ ويُقبِّلُهُ ويضعُهُ على يمينِ المذبحِ (أي جهةِ بحري) في لفافةِ حريرٍ.[^19] ويستبرئُ حالَ الخمرِ جيدًا بالشمِّ أو يدعو أحدَ المُفطرينَ يذوقُ منهُ في كفِّهِ، فإنَّ عوارضَ الخمرِ أكثرُ مِنْ عوارضِ القربانِ، فإنَّ القربانَ إذا حصلَ فيهِ عارضٌ ولم يوجدْ غيرُهُ فإنَّهُ لم يخرجْ عن طبعِ الخبزِ، لكنَّ الخمرَ إذا حصلَ فيهِ عارضٌ فسدَ، أي أنَّ الخمرَ إذا مالَ صارَ خلاًّ ولم يصلُحْ لهذهِ الخدمةِ، فيجبُ الاحترازُ مِنْ ذلكَ، ثُمَّ يُعطيهِ للشمَّاسِ فيمسكُهُ باللفافةِ.[^20]

! وإذا أكملَ الكاهنُ الدورةَ السابقَ ذكرُها يقفُ مكانَهُ أولاً غربيَّ المذبحِ ووجهُهُ إلى الشرقِ، ويضعُ الحملَ أي القربانةَ على يدِهِ اليُسرى، ويطامنُ رأسَهُ لإخوتِهِ الكهنةِ ويقولُ ⲉⲩⲗⲟⲅⲓⲧⲉ أي باركوا، وفي المفردِ إذا كانَ الحاضرُ عندَهُ كاهنًا واحدًا يقولُ ⲉⲩⲗⲟⲅⲏⲥⲟⲛ أي بارِكْ، فيطامنونَ هم أيضًا رؤوسَهم ويُجاوبونَهُ قائلينَ ⲛ̀ⲑⲟⲕ ⲉⲩⲗⲟⲅⲏⲥⲟⲛ أي أنتَ بارِكْ، ثُمَّ إنَّ الكاهنَ يلتفتُ إلى القربانةِ ويُقرِّبُ إليها وعاءَ الخمرِ ويكونُ الشمَّاسُ مُمسكًا لها بيدِهِ اليُمنى بلفافةِ حريرٍ،[^41] ويرشُمُ الكاهنُ الاثنينِ معًا ثلاثةَ رشومٍ بمثالِ الصليبِ كالعادةِ،[^42] فأوَّلُ ذلكَ يبتدئُ قائلاً
c: Ϧⲉⲛ ⲫ̀ⲣⲁⲛ ⲙ̀Ⲫⲓⲱⲧ ⲛⲉⲙ Ⲡϣⲏⲣⲓ ⲛⲉⲙ Ⲡⲓⲡ̀ⲛⲉⲩⲙⲁ ⲉⲑⲟⲩⲁⲃ ⲟⲩⲛⲟⲩϯ ⲛ̀ⲟⲩⲱⲧ.
a: باسمِ الآبِ والابنِ والروحِ القُدُسِ الإلهِ الواحدِ.
! ثُمَّ يرشُمُ الرشمَ الأولَ وهو يقولُ
c: Ⲥⲙⲁⲣⲱⲟⲩⲧ ⲛ̀ϫⲉ Ⲫϯ Ⲫⲓⲱⲧ ⲡⲓⲡⲁⲛⲧⲟⲕⲣⲁⲧⲱⲣ : ⲁⲙⲏⲛ.
a: مُبارَكٌ اللهُ الآبُ ضابطُ الكُلِّ، آمين.

# 229
! ويرشُمُ الرشمَ الثاني وهو يقولُ
c: Ⲥⲙⲁⲣⲱⲟⲩⲧ ⲛ̀ϫⲉ ⲡⲉϥⲙⲟⲛⲟⲅⲉⲛⲏⲥ ⲛ̀ϣⲏⲣⲓ Ⲓⲏⲥ Ⲡⲭⲥ Ⲡⲉⲛϭⲥ : ⲁⲙⲏⲛ.
a: مُبارَكٌ ابنُهُ الوحيدُ يسوعُ المسيحُ ربُّنا، آمين.
! ويرشُمُ الرشمَ الثالثَ وهو يقولُ
c: Ⲥⲙⲁⲣⲱⲟⲩⲧ ⲛ̀ϫⲉ Ⲡⲓⲡ̀ⲛⲉⲩⲙⲁ ⲉⲑⲟⲩⲁⲃ ⲙ̀ⲡⲁⲣⲁⲕⲗⲏⲧⲟⲛ : ⲁⲙⲏⲛ.
a: مُبارَكٌ الروحُ القُدُسُ المُعزِّي، آمين.
! ثُمَّ يضعُ القربانةَ في الصينيةِ وتحتَها لفافةُ حريرٍ وهو يقولُ
c: Ⲟⲩⲱⲟⲩ ⲛⲉⲙ ⲟⲩⲧⲁⲓⲟ : ⲟⲩⲧⲁⲓⲟ ⲛⲉⲙ ⲟⲩⲱⲟⲩ : ⲛ̀ϯⲡⲁⲛⲁⲅⲓⲁ ⲧⲣⲓⲁⲥ Ⲫⲓⲱⲧ ⲛⲉⲙ Ⲡϣⲏⲣⲓ ⲛⲉⲙ Ⲡⲓⲡ̀ⲛⲉⲩⲙⲁ ⲉⲑⲟⲩⲁⲃ.
a: مجدًا وكرامةً، كرامةً ومجدًا، للثالوثِ المُقدَّسِ الآبِ والابنِ والروحِ القُدُسِ.
c: Ϯⲛⲟⲩ ⲛⲉⲙ ⲛ̀ⲥⲏⲟⲩ ⲛⲓⲃⲉⲛ ⲛⲉⲙ ϣⲁ ⲉⲛⲉϩ ⲛ̀ⲧⲉ ⲛⲓⲉⲛⲉϩ ⲧⲏⲣⲟⲩ : ⲁⲙⲏⲛ.
a: الآنَ[^43] وكُلَّ أوانٍ وإلى دهرِ الداهرينَ كُلِّها، آمين.

# 230
! ثُمَّ يصبُّ الخمرَ في الكأسِ، ويمزجُها بقليلٍ مِنَ الماءِ نحوَ الثلثِ أو الرُّبعِ أو أقلَّ، ولا يزدْ عنِ الثلثِ.[^44]
! أمَّا الشمَّاسُ فإنَّهُ في كلِّ رشمٍ مِنَ الثلاثةِ رشوماتِ السابقةِ يقولُ ⲁⲙⲏⲛ وبعدَ ذلكَ يقولُ هذهِ التي في آخرِها المزمورُ ١١٦
c: Ⲓⲥ Ⲡⲁⲧⲏⲣ ⲁⲅⲓⲟⲥ : ⲓⲥ Ⲩⲓⲟⲥ ⲁⲅⲓⲟⲥ : ⲉⲛ Ⲡⲛⲉⲩⲙⲁ ⲁⲅⲓⲟⲛ ⲁⲙⲏⲛ : ⲉⲩⲗⲟⲅⲏⲧⲟⲥ Ⲕⲩⲣⲓⲟⲥ ⲟ Ⲑⲉⲟⲥ ⲓⲥ ⲧⲟⲩⲥ ⲉⲱⲛⲁⲥ : ⲁⲙⲏⲛ.
a: واحدٌ هو الآبُ القُدُّوسُ، واحدٌ هو الابنُ القُدُّوسُ، واحدٌ هو الروحُ القُدُسُ، آمين، مُبارَكٌ الربُّ الإلهُ إلى الأبدِ، آمين.
c: Ⲛⲓⲉⲑⲛⲟⲥ ⲧⲏⲣⲟⲩ ⲥⲙⲟⲩ ⲉ̀Ⲡϭⲥ : ⲙⲁⲣⲟⲩⲥⲙⲟⲩ ⲉⲣⲟϥ ⲛ̀ϫⲉ ⲛⲓⲗⲁⲟⲥ ⲧⲏⲣⲟⲩ : ϫⲉ ⲁⲡⲉϥⲛⲁⲓ ⲧⲁϫⲣⲟ ⲉϩⲣⲏⲓ ⲉϫⲱⲛ : ⲟⲩⲟϩ ϯⲙⲉⲑⲙⲏⲓ ⲛ̀ⲧⲉ Ⲡϭⲥ ϣⲟⲡ ϣⲁ ⲉⲛⲉϩ : ⲁⲙⲏⲛ ⲁⲗⲗⲏⲗⲟⲩⲓⲁ.
a: يا جميعَ الأممِ باركوا الربَّ ولتُبارِكْهُ جميعُ الشعوبِ، لأنَّ رحمتَهُ ثبتتْ علينا وحقُّ الربِّ يدومُ إلى الأبدِ، آمين، هللويا.

# 231
! يقولُ الشعبُ
c: Ⲇⲟⲝⲁ Ⲡⲁⲧⲣⲓ ⲕⲉ Ⲩ̀ⲓⲱ ⲕⲉ ⲁⲅⲓⲱ Ⲡⲛⲉⲩⲙⲁⲧⲓ : ⲕⲉ ⲛⲩⲛ ⲕⲉ ⲁⲓ ⲕⲉ ⲓⲥ ⲧⲟⲩⲥ ⲉⲱⲛⲁⲥ ⲧⲱⲛ ⲉⲱⲛⲱⲛ : ⲁⲙⲏⲛ ⲁⲗⲗⲏⲗⲟⲩⲓⲁ.
a: المجدُ للآبِ والابنِ والروحِ القُدُسِ، الآنَ وكُلَّ أوانٍ وإلى دهرِ الداهرينَ، آمين هللويا.
! ويُصفِّي قارورةَ الخمرِ جيدًا إلى الغايةِ ويمسحُ فمَها بلفافةٍ بيضاءَ ويُنزِلُها مِنَ المذبحِ. وبعدَ قراءةِ الشعبِ ⲇⲟⲝⲁ Ⲡⲁⲧⲣⲓ كما سبقَ.
